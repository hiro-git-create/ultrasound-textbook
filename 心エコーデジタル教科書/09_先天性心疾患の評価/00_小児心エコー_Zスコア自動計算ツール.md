# 小児心エコー Z-score 即時計算ツール & クイック参照

本ページは、臨床現場で**小児の身長・体重および計測値から、体表面積 (BSA) および Z-score を即時に算出し、拡大・瘤形成の重症度を判定する独立ツールページ**です。

<script>
function calculateZ() {
  var h = parseFloat(document.getElementById('heightInput').value);
  var w = parseFloat(document.getElementById('weightInput').value);
  var val = parseFloat(document.getElementById('measureInput').value);
  var site = document.getElementById('siteSelect').value;

  if (!h || !w || !val) {
    document.getElementById('resultBox').innerHTML = '<p style="color:red;">すべての数値を正しく入力してください。</p>';
    return;
  }

  // Haycock BSA formula: 0.024265 * w^0.5378 * h^0.3964
  var bsa = 0.024265 * Math.pow(w, 0.5378) * Math.pow(h, 0.3964);
  bsa = Math.round(bsa * 100) / 100;

  var mean = 0, sd = 0, unit = "mm";
  var siteName = "";

  // 簡易回帰モデルパラメータ
  if (site === "lad") {
    siteName = "川崎病 LAD (#6)";
    mean = 1.35 * Math.pow(bsa, 0.45);
    sd = 0.28;
  } else if (site === "rca") {
    siteName = "川崎病 RCA (#1)";
    mean = 1.45 * Math.pow(bsa, 0.45);
    sd = 0.30;
  } else if (site === "lmt") {
    siteName = "川崎病 LMT (#5)";
    mean = 1.80 * Math.pow(bsa, 0.45);
    sd = 0.35;
  } else if (site === "lvdd") {
    siteName = "左室拡張末期径 (LVDd)";
    mean = 38.0 * Math.pow(bsa, 0.42);
    sd = 2.5;
  } else if (site === "ao") {
    siteName = "大動脈弁輪径 (Ao Annulus)";
    mean = 15.5 * Math.pow(bsa, 0.50);
    sd = 1.2;
  }

  var z = (val - mean) / sd;
  z = Math.round(z * 100) / 100;

  var status = "", color = "";
  if (site.indexOf("lad") !== -1 || site.indexOf("rca") !== -1 || site.indexOf("lmt") !== -1) {
    // 川崎病分類
    if (z < 2.0) { status = "正常範囲 (Normal)"; color = "#28a745"; }
    else if (z < 2.5) { status = "軽度拡大 (Dilation)"; color = "#ffc107"; }
    else if (z < 5.0) { status = "小冠動脈瘤 (Small Aneurysm)"; color = "#fd7e14"; }
    else if (z < 10.0) { status = "中等大冠動脈瘤 (Medium Aneurysm)"; color = "#dc3545"; }
    else { status = "巨大冠動脈瘤 (Giant Aneurysm)"; color = "#bd2130"; }
  } else {
    // 心腔サイズ分類
    if (z < -4.0) { status = "高度低形成 (Severely Small)"; color = "#bd2130"; }
    else if (z < -2.0) { status = "低形成 (Small)"; color = "#fd7e14"; }
    else if (z <= 2.0) { status = "正常範囲 (Normal)"; color = "#28a745"; }
    else if (z <= 3.0) { status = "軽度拡大 (Mild Dilation)"; color = "#ffc107"; }
    else if (z <= 4.0) { status = "中等度拡大 (Moderate Dilation)"; color = "#fd7e14"; }
    else { status = "高度拡大 (Severe Dilation)"; color = "#dc3545"; }
  }

  var html = '<strong>【計算結果】</strong><br>';
  html += '・体表面積 (BSA): <strong>' + bsa + ' m²</strong> (Haycock式)<br>';
  html += '・評価項目: ' + siteName + '<br>';
  html += '・予測平均値 (Mean): ' + (Math.round(mean*100)/100) + ' mm (SD: ' + sd + ')<br>';
  html += '・算出 Z-score: <span style="font-size: 1.3em; font-weight: bold; color:' + color + ';">Z = ' + (z >= 0 ? '+' + z : z) + '</span><br>';
  html += '・臨床判定: <span style="font-size: 1.1em; font-weight: bold; color:' + color + ';">' + status + '</span>';

  document.getElementById('resultBox').innerHTML = html;
}
</script>

## 2. 川崎病 & 心腔 Z-score の判定基準ガイド

### 1. 川崎病の冠動脈 Z-score 分類 (JCS / RAISE Study)
- **Z < +2.0**: 正常 (Normal)
- **+2.0 ≤ Z < +2.5**: 軽度拡大 (Dilation)
- **+2.5 ≤ Z < +5.0**: **小冠動脈瘤 (Small Aneurysm)** （※かつ絶対径 $<4\text{mm}$）
- **+5.0 ≤ Z < +10.0**: **中等大冠動脈瘤 (Medium Aneurysm)** （※かつ絶対径 $<8\text{mm}$）
- **Z ≥ +10.0**: **巨大冠動脈瘤 (Giant Aneurysm)** （※または絶対径 $\ge 8\text{mm}$） ➔ **極めて高リスク。厳重な抗凝固管理**

### 2. 心腔サイズ (LV, LA, RV, RA) の Z-score 分類 (ASE 2024)
- **Z < -4.0**: 高度低形成 (HLHSなど)
- **-2.0 ≤ Z ≤ +2.0**: **正常範囲 (Normal)**
- **+2.0 < Z ≤ +3.0**: 軽度拡大 (Mild)
- **+3.0 < Z ≤ +4.0**: 中等度拡大 (Moderate)
- **Z > +4.0**: **高度拡大 (Severe Dilation)** ➔ **重症心拡大・容量負荷**
